# 가장 많이 받은 선물
### [Lv.1 - 258712](https://school.programmers.co.kr/learn/courses/30/lessons/258712)

## 문제
선물을 직접 전하기 힘들 때 카카오톡 선물하기 기능을 이용해 축하 선물을 보낼 수 있습니다. 당신의 친구들이 이번 달까지 선물을 주고받은 기록을 바탕으로 다음 달에 누가 선물을 많이 받을지 예측하려고 합니다.

- 두 사람이 선물을 주고받은 기록이 있다면, 이번 달까지 두 사람 사이에 더 많은 선물을 준 사람이 다음 달에 선물을 하나 받습니다.
  - 예를 들어 A가 B에게 선물을 5번 줬고, B가 A에게 선물을 3번 줬다면 다음 달엔 A가 B에게 선물을 하나 받습니다.
- 두 사람이 선물을 주고받은 기록이 하나도 없거나 주고받은 수가 같다면, 선물 지수가 더 큰 사람이 선물 지수가 더 작은 사람에게 선물을 하나 받습니다.
  - 선물 지수는 이번 달까지 자신이 친구들에게 준 선물의 수에서 받은 선물의 수를 뺀 값입니다.
  - 예를 들어 A가 친구들에게 준 선물이 3개고 받은 선물이 10개라면 A의 선물 지수는 -7입니다. B가 친구들에게 준 선물이 3개고 받은 선물이 2개라면 B의 선물 지수는 1입니다. 만약 A와 B가 선물을 주고받은 적이 없거나 정확히 같은 수로 선물을 주고받았다면, 다음 달엔 B가 A에게 선물을 하나 받습니다.
  - 만약 두 사람의 선물 지수도 같다면 다음 달에 선물을 주고받지 않습니다.

위에서 설명한 규칙대로 다음 달에 선물을 주고받을 때, 당신은 선물을 가장 많이 받을 친구가 받을 선물의 수를 알고 싶습니다.
친구들의 이름을 담은 1차원 문자열 배열 `friends` 이번 달까지 친구들이 주고받은 선물 기록을 담은 1차원 문자열 배열 `gifts`가 매개변수로 주어집니다. 
이때, 다음달에 가장 많은 선물을 받는 친구가 받을 선물의 수를 return 하도록 solution 함수를 완성해 주세요.
## 풀이
- 선물을 서로 주고 받은 기록과 횟수까지 기억해야 하기에 검색하기 용이하도록 `Dictionary<string, Dictionary<string, int>>` 같은 형태로 값을 저장했다.

``` C#
using System;
using System.Collections.Generic;

public class Solution {
    public int solution(string[] friends, string[] gifts) {
        // [선물을 준 대상자, [선물을 받은 대상자, 선물 받은 횟수]]
        Dictionary<string, Dictionary<string, int>> giftInfo = new Dictionary<string, Dictionary<string, int>>();
        // 선물 지수
        Dictionary<string, int> giftPoint = new Dictionary<string, int>();
        // 다음 달 선물 정보[선물 받는 대상자, 선물 개수]
        Dictionary<string, int> nextMonthGift = new Dictionary<string, int>();
        int answer = 0;
        
        
        // 선물 정보 초기화
        foreach(string give in friends)
        {
            giftInfo[give] = new Dictionary<string, int>();
            
            // 선물 횟수 초기화
            foreach(string take in friends)
            {
                // 선물을 주고 받을 때 동일인은 제외
                if(give == take)    continue;
                giftInfo[give][take] = 0;
            }
        }
        
        // 다음 달 선물 초기화
        foreach(string friend in friends) nextMonthGift[friend] = 0;
        
        // 서로에게 선물 준 카운트 정리
        for(int i = 0; i < gifts.Length; i++)
        {
            // [0] : give, [1] : take
            string[] user = gifts[i].Split(' ');
            // 선물 주고 받음 카운트
            giftInfo[user[0]][user[1]]++;
        }
        
        // 선물 지수 계산하기
        foreach(string user in friends)
        {
            // 선물 준 횟수 / 선물 받은 횟수
            int giveCount = 0;
            int takeCount = 0;

            // 자신이 준 선물 횟수 총합
            foreach(int count in giftInfo[user].Values) 
            {
                giveCount += count;
            }

            // 자신이 받은 선물 횟수 총합
            foreach(string other in friends) 
            {
                if (user == other) continue;
                takeCount += giftInfo[other][user];
            }
            
            // 선물 지수 계산
            giftPoint[user] = giveCount - takeCount;
        }
        
        
        for(int i = 0; i < friends.Length; i++)
        {
            for(int j = i + 1; j < friends.Length; j++) // 중복 방지를 위해 i+1부터 시작
            {
                // 서로를 A와 B로 임시 정의
                string A = friends[i];
                string B = friends[j];
                
                // 서로 주고받은 선물 정보 불러오기
                int aToB = giftInfo[A][B]; 
                int bToA = giftInfo[B][A]; 
                
                // 횟수에 따라 다음 달 선물 횟수에 추가
                if(aToB > bToA) 
                    nextMonthGift[A]++;
                else if(bToA > aToB)
                    nextMonthGift[B]++;
                // 서로 주고 받은 횟수가 동일하면
                else 
                {
                    // 선물지수로 비교
                    if(giftPoint[A] > giftPoint[B]) 
                        nextMonthGift[A]++;
                    else if(giftPoint[B] > giftPoint[A]) 
                        nextMonthGift[B]++;
                }
            }   
        }
        
        // 선물을 가장 많이 받는 자 검색
        foreach(int count in nextMonthGift.Values)
        {
            if(count > answer) 
                answer = count;
        }
        return answer;
    }
}
```
